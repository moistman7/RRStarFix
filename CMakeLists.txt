cmake_minimum_required(VERSION 3.24)
project(RRStarFix VERSION 0.9.1 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with the Windows x64 MSVC toolchain")
endif()
find_program(RRSTARFIX_DXC NAMES dxc.exe dxc REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/star_composite_shader.hpp"
  COMMAND "${RRSTARFIX_DXC}" -T cs_6_0 -E cs_main -O3 -Ges -HV 2021
    -Fo "${CMAKE_CURRENT_BINARY_DIR}/star_composite.dxil" "${CMAKE_CURRENT_SOURCE_DIR}/src/star_composite.hlsl"
  COMMAND "${CMAKE_COMMAND}" "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/star_composite.dxil" -DNAME=star_composite_shader
    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/star_composite_shader.hpp" -P "${CMAKE_CURRENT_SOURCE_DIR}/src/embed.cmake"
  DEPENDS src/star_composite.hlsl src/embed.cmake VERBATIM)
add_custom_target(shaders DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/star_composite_shader.hpp")
configure_file(src/version.rc.in "${CMAKE_CURRENT_BINARY_DIR}/version.rc" @ONLY)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(RRStarFix SHARED src/addon.cpp src/sl_probe.cpp src/gpu_snapshot.cpp src/star_composite.cpp src/config.cpp
  "${CMAKE_CURRENT_BINARY_DIR}/version.rc")
add_dependencies(RRStarFix shaders)
target_compile_features(RRStarFix PRIVATE cxx_std_20)
target_include_directories(RRStarFix PRIVATE third_party/reshade third_party/streamline "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(RRStarFix PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX RRSTARFIX_VERSION="${PROJECT_VERSION}")
target_compile_options(RRStarFix PRIVATE /W4 /EHsc /wd4996)
target_link_libraries(RRStarFix PRIVATE user32 minhook d3d12 dxgi)
set_target_properties(RRStarFix PROPERTIES PREFIX "" SUFFIX ".addon64")
include(CTest)
